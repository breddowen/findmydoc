@dataclass(frozen=True)
class SuperuserAgentConfig:
    """
    Настройки графа суперпользователя.
    Здесь можно независимо менять модель
    каждого узла между PRIMARY и SECONDARY.
    """
    orchestrator_history_message_limit: int = 12
    
    orchestrator_history_token_limit: int = 2500
    agent_history_token_limit: int = 2000
    general_history_message_limit: int = 2
    current_patient_history_message_limit: int = 2
    web_research_history_message_limit: int = 2
    memory_history_message_limit: int = 2
    general_memory_limit: int = 8
    recent_tool_results_limit: int = 4
    recent_tool_result_max_chars: int = 4000
    handoff_max_chars: int = 3000
    web_source_registry_limit: int = 20
    max_tool_iterations: int = 8
    # На первом этапе все узлы работают
    # через PRIMARY.
    #
    # Для эксперимента достаточно заменить
    # роль нужного узла на SECONDARY.
    orchestrator_model_role: AIModelRole = (
        AIModelRole.PRIMARY
    )
    general_model_role: AIModelRole = (
        AIModelRole.PRIMARY
    )
    current_patient_model_role: AIModelRole = (
        AIModelRole.PRIMARY
    )
    web_research_model_role: AIModelRole = (
        AIModelRole.PRIMARY
    )
    memory_model_role: AIModelRole = (
        AIModelRole.PRIMARY
    )
@dataclass(frozen=True)
class SuperuserAgentModels:
    """
    Рабочие экземпляры LangChain-моделей.
    """
    primary: Any
    secondary: Any
    def resolve(
        self,
        role: AIModelRole
    ):
        if role == AIModelRole.PRIMARY:
            return self.primary
        if role == AIModelRole.SECONDARY:
            return self.secondary
        raise ValueError(
            f"Unsupported model role: {role}"
        )

WEB_TOOL_NAMES = {
    "web_search",
    "search_medical_sources",
    "scrape_web_page",
    "list_available_medical_sources"
}

def load_conversation_messages(
    runtime_context: AgentRuntimeContext,
    limit: int,
    token_budget: int,
) -> list:
    with (
        runtime_context
        .session_factory()
    ) as session:
        statement = select(
            AIChatMessage
        ).where(
            AIChatMessage.chat_id
            == runtime_context.chat_id,
            AIChatMessage.status
            == ChatMessageStatus.COMPLETED,
        )

        context_changed_at = (
            runtime_context
            .chat_context_changed_at
        )

        if context_changed_at:
            statement = statement.where(
                AIChatMessage.created_at
                >= context_changed_at
            )

        statement = statement.order_by(
            AIChatMessage.sequence.desc()
        ).limit(
            max(limit, 1)
        )

        records = list(
            session.exec(
                statement
            ).all()
        )

        records.reverse()

        messages = []

        for record in records:
            if (
                record.role
                == ChatMessageRole.USER
            ):
                messages.append(
                    HumanMessage(
                        content=record.content,
                        id=record.id,
                    )
                )

            elif (
                record.role
                == ChatMessageRole.ASSISTANT
            ):
                messages.append(
                    AIMessage(
                        content=record.content,
                        id=record.id,
                    )
                )

        return trim_conversation_messages(
            messages=messages,
            token_budget=token_budget,
        )


def load_recent_tool_results(
    runtime_context: AgentRuntimeContext,
    config: SuperuserAgentConfig,
) -> list[dict]:
    with (
        runtime_context
        .session_factory()
    ) as session:
        statement = select(
            AIChatMessage
        ).where(
            AIChatMessage.chat_id
            == runtime_context.chat_id,
            AIChatMessage.role
            == ChatMessageRole.ASSISTANT,
        )

        context_changed_at = (
            runtime_context
            .chat_context_changed_at
        )

        if context_changed_at:
            statement = statement.where(
                AIChatMessage.created_at
                >= context_changed_at
            )

        statement = statement.order_by(
            AIChatMessage.sequence.desc()
        ).limit(30)

        messages = list(
            session.exec(
                statement
            ).all()
        )

        results = []

        for message in messages:
            parts = get_message_parts(
                session=session,
                message_id=message.id,
            )

            parts.reverse()

            for part in parts:
                if (
                    part.part_type
                    != ChatMessagePartType
                    .TOOL_RESULT
                ):
                    continue

                results.append({
                    "message_id": message.id,
                    "tool_name": part.tool_name,
                    "tool_call_id": (
                        part.tool_call_id
                    ),
                    "content": truncate_text(
                        part.content,
                        config
                        .recent_tool_result_max_chars,
                    ),
                })

                if (
                    len(results)
                    >= config
                    .recent_tool_results_limit
                ):
                    return results

        return results


def extract_web_sources(
    recent_tool_results: list[dict],
    existing_sources: list[dict],
    limit: int
) -> list[dict]:
    by_url = {}

    for source in existing_sources:
        url = source.get("url")

        if url:
            by_url[url] = source

    # Идём от более старых результатов
    # к новым, чтобы новые значения могли
    # обновить заголовки и snippets.
    for tool_result in reversed(
        recent_tool_results
    ):
        if (
            tool_result.get("tool_name")
            not in WEB_TOOL_NAMES
        ):
            continue

        content = tool_result.get(
            "content"
        )

        if not content:
            continue

        try:
            payload = json.loads(
                content
            )
        except (
            json.JSONDecodeError,
            TypeError
        ):
            continue

        items = []

        if isinstance(payload, dict):
            raw_results = payload.get(
                "results"
            )

            if isinstance(
                raw_results,
                list
            ):
                items.extend(
                    raw_results
                )

            if payload.get("url"):
                items.append(payload)

        elif isinstance(payload, list):
            items.extend(payload)

        for item in items:
            if not isinstance(item, dict):
                continue

            url = (
                item.get("url")
                or item.get("href")
                or item.get("link")
            )

            if not url:
                continue

            parsed_url = urlparse(url)

            by_url[url] = {
                "title": (
                    item.get("title")
                    or item.get("name")
                    or url
                ),
                "url": url,
                "domain": (
                    parsed_url.netloc
                    or None
                ),
                "source": (
                    item.get("source")
                    or item.get(
                        "source_name"
                    )
                ),
                "snippet": truncate_text(
                    (
                        item.get("snippet")
                        or item.get(
                            "description"
                        )
                        or item.get(
                            "content"
                        )
                    ),
                    1000
                )
            }

    sources = list(
        by_url.values()
    )[-limit:]

    for index, source in enumerate(
        sources,
        start=1
    ):
        source["number"] = index

    return sources


def load_memory_items(
    runtime_context: AgentRuntimeContext,
    limit: int,
) -> list[dict]:
    with (
        runtime_context
        .session_factory()
    ) as session:
        memories = list_user_memories(
            session=session,
            subject_user_id=(
                runtime_context
                .current_user_id
            ),
            include_inactive=False,
            limit=limit,
        )

        return [
            {
                "id": memory.id,
                "category": (
                    memory.category.value
                ),
                "content": memory.content,
                "importance": (
                    memory.importance
                ),
                "confidence": (
                    memory.confidence
                ),
            }
            for memory in memories
        ]


def build_prepare_context_node(
    config: SuperuserAgentConfig
):
    def prepare_context(
        state,
        runtime: Runtime[
            AgentRuntimeContext
        ]
    ) -> dict:
        runtime_context = (
            runtime.context
        )

        conversation_messages = (
            load_conversation_messages(
                runtime_context=(
                    runtime_context
                ),
                limit=(
                    config
                    .orchestrator_history_message_limit
                ),
                token_budget=(
                    config
                    .orchestrator_history_token_limit
                )
            )
        )

        recent_tool_results = (
            load_recent_tool_results(
                runtime_context=(
                    runtime_context
                ),
                config=config
            )
        )

        memory_items = (
            load_memory_items(
                runtime_context=(
                    runtime_context
                ),
                limit=(
                    config
                    .general_memory_limit
                )
            )
        )

        existing_sources = (
            state.get(
                "web_sources",
                []
            )
            or []
        )

        web_sources = extract_web_sources(
            recent_tool_results=(
                recent_tool_results
            ),
            existing_sources=(
                existing_sources
            ),
            limit=(
                config
                .web_source_registry_limit
            )
        )

        # chat = runtime_context.chat

        return {
            "conversation_messages": (
                conversation_messages
            ),
            "memory_items": memory_items,
            "recent_tool_results": (
                recent_tool_results
            ),
            "web_sources": web_sources,
            "patient_context_active": (
                runtime_context
                .chat_context_mode
                == "patient"
                and bool(
                    runtime_context
                    .chat_context_patient_id
                )
            ),
            "patient_profile_id": (
                runtime_context
                .chat_context_patient_id
            ),
            "final_response": "",
            "messages": [],
            "tool_iterations": 0
        }

    return prepare_context

def build_current_patient_graph(
    model,
    config: SuperuserAgentConfig
):
    tools = [
        get_current_patient_summary,
        get_current_patient_facts,
        search_current_patient_facts,
        propose_current_patient_facts,
        propose_retract_current_patient_facts
    ]

    builder = StateGraph(
        ToolAgentState,
        context_schema=AgentRuntimeContext
    )

    builder.add_node(
        "initialize",
        initialize_agent_messages
    )

    builder.add_node(
        "model",
        build_model_node(
            model=model,
            system_prompt=(
                CURRENT_PATIENT_AGENT_PROMPT
            ),
            tools=tools
        )
    )

    builder.add_node(
        "tools",
        ToolNode(
            tools=tools,
            messages_key=(
                "agent_messages"
            ),
            handle_tool_errors=True
        )
    )

    builder.add_node(
        "increment_tool_iterations",
        increment_tool_iterations
    )

    builder.add_edge(
        START,
        "initialize"
    )

    builder.add_edge(
        "initialize",
        "model"
    )

    builder.add_conditional_edges(
        "model",
        build_after_model_router(
            config.max_tool_iterations
        ),
        {
            "tools": "tools",
            "end": END
        }
    )

    builder.add_edge(
        "tools",
        "increment_tool_iterations"
    )

    builder.add_edge(
        "increment_tool_iterations",
        "model"
    )

    return builder.compile(
        checkpointer=None
    )

class SuperuserAgentRoute(
    str,
    Enum
):
    GENERAL = "general"

    CURRENT_PATIENT = (
        "current_patient"
    )

    WEB_RESEARCH = (
        "web_research"
    )

    MEMORY = "memory"

    CLARIFY = "clarify"


AGENT_ROUTE_TITLES = {
    SuperuserAgentRoute.GENERAL: (
        "Обычный медицинский диалог"
    ),
    SuperuserAgentRoute.CURRENT_PATIENT: (
        "Работа с выбранным пациентом"
    ),
    SuperuserAgentRoute.WEB_RESEARCH: (
        "Поиск медицинской информации"
    ),
    SuperuserAgentRoute.MEMORY: (
        "Работа с долговременной памятью"
    ),
    SuperuserAgentRoute.CLARIFY: (
        "Уточняю запрос"
    )
}


def normalize_agent_route(
    value
) -> SuperuserAgentRoute:
    if isinstance(
        value,
        SuperuserAgentRoute
    ):
        return value

    return SuperuserAgentRoute(
        str(value)
    )


def get_agent_route_title(
    route
) -> str:
    normalized = normalize_agent_route(
        route
    )

    return AGENT_ROUTE_TITLES[
        normalized
    ]

def build_general_graph(
    model,
    config: SuperuserAgentConfig,
    patient_context: bool
):
    # В активном patient context общий
    # агент не должен искать других
    # пациентов.
    tools = (
        []
        if patient_context
        else [
            list_patients,
            find_patients,
            get_patient_summary
        ]
    )

    builder = StateGraph(
        ToolAgentState,
        context_schema=AgentRuntimeContext
    )

    builder.add_node(
        "initialize",
        initialize_agent_messages
    )

    builder.add_node(
        "model",
        build_model_node(
            model=model,
            system_prompt=(
                GENERAL_AGENT_PROMPT
            ),
            tools=tools
        )
    )

    builder.add_edge(
        START,
        "initialize"
    )

    builder.add_edge(
        "initialize",
        "model"
    )

    if not tools:
        builder.add_edge(
            "model",
            END
        )

        return builder.compile(
            checkpointer=None
        )

    builder.add_node(
        "tools",
        ToolNode(
            tools=tools,
            messages_key=(
                "agent_messages"
            ),
            handle_tool_errors=True
        )
    )

    builder.add_node(
        "increment_tool_iterations",
        increment_tool_iterations
    )

    builder.add_conditional_edges(
        "model",
        build_after_model_router(
            config.max_tool_iterations
        ),
        {
            "tools": "tools",
            "end": END
        }
    )

    builder.add_edge(
        "tools",
        "increment_tool_iterations"
    )

    builder.add_edge(
        "increment_tool_iterations",
        "model"
    )

    return builder.compile(
        checkpointer=None
    )

def format_memory_context(
    memory_items: list[dict]
) -> str:
    if not memory_items:
        return ""

    lines = [
        "Долговременная память о текущем "
        "пользователе:"
    ]

    for item in memory_items:
        lines.append(
            "- "
            f"[{item.get('category')}] "
            f"{item.get('content')}"
        )

    lines.append(
        "\nИспользуй эти сведения только "
        "как предпочтения и контекст общения. "
        "Не представляй их как медицинские "
        "данные пациента."
    )

    return "\n".join(lines)


def format_patient_context(
    state
) -> str:
    if not state.get(
        "patient_context_active"
    ):
        return ""

    patient_name = (
        state.get("patient_name")
        or "Имя не указано"
    )

    record_id = (
        state.get(
            "patient_record_id"
        )
        or "не указан"
    )

    return (
        "Активный контекст пациента:\n"
        f"- пациент: {patient_name};\n"
        f"- номер карты: {record_id}.\n\n"
        "Работай только с этим пациентом. "
        "Не смешивай его данные с данными "
        "других пациентов."
    )


def format_web_sources(
    sources: list[dict]
) -> str:
    if not sources:
        return ""

    lines = [
        "Источники из предыдущих "
        "веб-запросов:"
    ]

    for source in sources:
        number = source.get(
            "number"
        )

        title = (
            source.get("title")
            or source.get("url")
        )

        url = source.get("url")

        snippet = source.get(
            "snippet"
        )

        line = (
            f"{number}. {title} — {url}"
        )

        if snippet:
            line += (
                f"\n   Краткое описание: "
                f"{snippet}"
            )

        lines.append(line)

    lines.append(
        "\nЕсли пользователь просит открыть "
        "первый, второй или другой источник, "
        "используй соответствующий URL из "
        "этого списка."
    )

    return "\n".join(lines)


def build_prepare_agent_input_node(
    config: SuperuserAgentConfig
):
    def prepare_agent_input(
        state
    ) -> dict:
        route = normalize_agent_route(
            state.get(
                "route",
                SuperuserAgentRoute.GENERAL
            )
        )

        history_limits = {
            SuperuserAgentRoute.GENERAL: (
                config
                .general_history_message_limit
            ),
            SuperuserAgentRoute
            .CURRENT_PATIENT: (
                config
                .current_patient_history_message_limit
            ),
            SuperuserAgentRoute
            .WEB_RESEARCH: (
                config
                .web_research_history_message_limit
            ),
            SuperuserAgentRoute.MEMORY: (
                config
                .memory_history_message_limit
            ),
            SuperuserAgentRoute.CLARIFY: 1
        }

        history_limit = history_limits[
            route
        ]

        history = list(
            state.get(
                "conversation_messages",
                []
            )
        )[-history_limit:]

        history = trim_conversation_messages(
            messages=history,
            token_budget=(
                config.agent_history_token_limit
            )
        )

        context_messages = []

        if (
            route
            == SuperuserAgentRoute.GENERAL
        ):
            memory_context = (
                format_memory_context(
                    state.get(
                        "memory_items",
                        []
                    )
                )
            )

            if memory_context:
                context_messages.append(
                    SystemMessage(
                        content=memory_context
                    )
                )

            if state.get(
                "patient_context_active"
            ):
                context_messages.append(
                    SystemMessage(
                        content=(
                            "В чате выбран пациент, "
                            "но текущий запрос "
                            "маршрутизирован как "
                            "обычный медицинский "
                            "диалог. Не ищи других "
                            "пациентов и не изменяй "
                            "медицинскую карту."
                        )
                    )
                )

        elif (
            route
            == SuperuserAgentRoute
            .CURRENT_PATIENT
        ):
            patient_context = (
                format_patient_context(
                    state
                )
            )

            if patient_context:
                context_messages.append(
                    SystemMessage(
                        content=(
                            patient_context
                        )
                    )
                )

        elif (
            route
            == SuperuserAgentRoute
            .WEB_RESEARCH
        ):
            web_context = (
                format_web_sources(
                    state.get(
                        "web_sources",
                        []
                    )
                )
            )

            if web_context:
                context_messages.append(
                    SystemMessage(
                        content=web_context
                    )
                )

            if state.get(
                "patient_context_active"
            ):
                context_messages.append(
                    SystemMessage(
                        content=(
                            "В чате выбран пациент. "
                            "Для внешнего поиска "
                            "используй только "
                            "обезличенные данные. "
                            "Не передавай ФИО, email, "
                            "номер карты, телефон "
                            "или адрес пациента."
                        )
                    )
                )

        handoff_context = truncate_text(
            state.get(
                "handoff_context"
            ),
            config.handoff_max_chars
        )

        # Handoff размещается непосредственно
        # перед текущим сообщением пользователя.
        if history:
            previous_messages = history[:-1]
            current_message = history[-1:]

        else:
            previous_messages = []
            current_message = []

        handoff_messages = []

        if handoff_context:
            handoff_messages.append(
                SystemMessage(
                    content=(
                        "Контекст передачи от "
                        "оркестратора:\n"
                        f"{handoff_context}"
                    )
                )
            )

        messages = [
            *context_messages,
            *previous_messages,
            *handoff_messages,
            *current_message
        ]

        return {
            "messages": messages,
            "tool_iterations": 0,
            "final_response": ""
        }

    return prepare_agent_input

        "final_response": response

def build_orchestrator_node(

        model.with_structured_output(
            OrchestratorDecision
        )
    )

    async def orchestrate(
        state
    ) -> dict:
        previous_route = state.get(
            "route"
        )

        metadata = {
            "patient_context_active": (
                state.get(
                    "patient_context_active",
                    False
                )
            ),
            "patient_name": (
                state.get("patient_name")
            ),
            "previous_route": (
                (
                    normalize_agent_route(
                        previous_route
                    ).value
                )
                if previous_route
                else None
            ),
            "recent_tool_results": (
                state.get(
                    "recent_tool_results",
                    []
                )
            ),
            "known_web_sources": (
                state.get(
                    "web_sources",
                    []
                )
            )
        }

        messages = [
            SystemMessage(
                content=(
                    ORCHESTRATOR_PROMPT
                )
            ),
            SystemMessage(
                content=(
                    "Технический контекст:\n"
                    + json.dumps(
                        metadata,
                        ensure_ascii=False,
                        indent=2
                    )
                )
            ),
            *state.get(
                "conversation_messages",
                []
            )
        ]

        decision = await (
            structured_model.ainvoke(
                messages
            )
        )

        if not isinstance(
            decision,
            OrchestratorDecision
        ):
            decision = (
                OrchestratorDecision
                .model_validate(
                    decision
                )
            )

        return {
            "previous_route": (
                normalize_agent_route(
                    previous_route
                )
                if previous_route
                else None
            ),
            "route": decision.route,
            "intent_summary": (
                decision.intent_summary
            ),
            "handoff_context": (
                decision.handoff_context
            ),
            "clarification_question": (
                decision
                .clarification_question
            )
        }

    return orchestrate

def normalize_tool_messages(
    messages: list,
) -> list:
    """
    Groq требует, чтобы content сообщения
    с role=tool был строкой либо непустым
    списком.

    Для совместимости преобразуем любой
    нестроковый результат инструмента
    в JSON-строку.
    """

    normalized_messages = []

    for message in messages:
        if not isinstance(
            message,
            ToolMessage,
        ):
            normalized_messages.append(
                message
            )
            continue

        content = message.content

        if isinstance(content, str):
            normalized_content = content

        elif content is None:
            normalized_content = "null"

        else:
            normalized_content = json.dumps(
                jsonable_encoder(content),
                ensure_ascii=False,
            )

        normalized_messages.append(
            message.model_copy(
                update={
                    "content": (
                        normalized_content
                    ),
                }
            )
        )

    return normalized_messages

def build_model_node(
    model,
    system_prompt: str,
    tools: Sequence | None = None,
) -> Callable:
    """
    Создать узел вызова модели.

    RunnableConfig передаётся во вложенный
    вызов модели, чтобы внешний LangGraph
    видел потоковые chunks.
    """

    tools = list(
        tools or []
    )

    model_with_tools = (
        model.bind_tools(tools)
        if tools
        else model
    )

    async def call_model(
        state,
        config: RunnableConfig,
    ) -> dict:
        messages = [
            SystemMessage(
                content=system_prompt
            ),
            *state.get(
                "agent_messages",
                [],
            ),
        ]

        model_messages = (
            normalize_tool_messages(
                messages
            )
        )

        inherited_tags = list(
            config.get(
                "tags",
                [],
            )
            or []
        )

        if (
            "agent_stream_output"
            not in inherited_tags
        ):
            inherited_tags.append(
                "agent_stream_output"
            )

        stream_config: RunnableConfig = {
            **config,
            "tags": inherited_tags,
        }

        response_chunk = None

        async for chunk in (
            model_with_tools.astream(
                model_messages,
                config=stream_config,
            )
        ):
            if response_chunk is None:
                response_chunk = chunk
            else:
                response_chunk = (
                    response_chunk + chunk
                )

        if response_chunk is None:
            raise RuntimeError(
                "Model stream finished "
                "without a response"
            )

        response = (
            message_chunk_to_message(
                response_chunk
            )
        )

        update = {
            "agent_messages": [
                response
            ],
        }

        tool_calls = getattr(
            response,
            "tool_calls",
            None,
        ) or []

        if not tool_calls:
            update["final_response"] = (
                normalize_message_content(
                    response.content
                )
            )

        return update

    return call_model


def build_after_model_router(
    max_tool_iterations: int
) -> Callable:
    def route_after_model(
        state
    ) -> str:
        messages = state.get(
            "agent_messages",
            []
        )

        if not messages:
            return "end"

        last_message = messages[-1]

        if not isinstance(
            last_message,
            AIMessage
        ):
            return "end"

        tool_calls = (
            last_message.tool_calls
            or []
        )

        if not tool_calls:
            return "end"

        tool_iterations = state.get(
            "tool_iterations",
            0
        )

        if (
            tool_iterations
            >= max_tool_iterations
        ):
            raise RuntimeError(
                "Agent exceeded the maximum "
                "number of tool iterations"
            )

        return "tools"

    return route_after_model


def increment_tool_iterations(
    state
) -> dict:
    return {
        "tool_iterations": (
            state.get(
                "tool_iterations",
                0
            )
            + 1
        )
    }

import logging
logger = logging.getLogger(
    __name__
)

import json

from fastapi.encoders import (
    jsonable_encoder
)
from langchain_core.messages import (
    AIMessage
)

from langchain_core.messages.utils import (
    count_tokens_approximately
)

def normalize_message_content(
    content
) -> str:
    if content is None:
        return ""

    if isinstance(content, str):
        return content.strip()

    if isinstance(content, list):
        values = []

        for item in content:
            if isinstance(item, str):
                values.append(item)
                continue

            if isinstance(item, dict):
                text = (
                    item.get("text")
                    or item.get("content")
                )

                if text:
                    values.append(
                        str(text)
                    )

        if values:
            return "".join(
                values
            ).strip()

    try:
        return json.dumps(
            jsonable_encoder(content),
            ensure_ascii=False
        )

    except Exception:
        return str(content).strip()


def truncate_text(
    value: str | None,
    max_chars: int
) -> str:
    if not value:
        return ""

    normalized = str(value).strip()

    if len(normalized) <= max_chars:
        return normalized

    return (
        normalized[:max_chars]
        + "\n...[контекст сокращён]"
    )

def trim_conversation_messages(
    messages: list,
    token_budget: int
) -> list:
    """
    Удаляет самые старые сообщения истории.

    System prompts здесь отсутствуют, поэтому
    функция обрезает исключительно сообщения
    диалога.

    Последнее сообщение пользователя всегда
    сохраняется, даже если оно само превышает
    бюджет.
    """

    if not messages:
        return []

    if token_budget <= 0:
        return [
            messages[-1]
        ]

    trimmed = list(messages)

    initial_tokens = (
        count_tokens_approximately(
            trimmed
        )
    )

    while (
        len(trimmed) > 1
        and count_tokens_approximately(
            trimmed
        ) > token_budget
    ):
        trimmed.pop(0)

        """
         После удаления HumanMessage может
         остаться ответ AI без вопроса.
         Такие сообщения тоже удаляем.
        """

        while (
            len(trimmed) > 1
            and isinstance(
                trimmed[0],
                AIMessage
            )
        ):
            trimmed.pop(0)

    final_tokens = (
        count_tokens_approximately(
            trimmed
        )
    )

    if len(trimmed) < len(messages):
        logger.info(
            "Conversation history trimmed: "
            "messages=%s->%s tokens=%s->%s "
            "budget=%s",
            len(messages),
            len(trimmed),
            initial_tokens,
            final_tokens,
            token_budget
        )

    if final_tokens > token_budget:
        logger.warning(
            "Latest conversation message "
            "exceeds token budget: "
            "tokens=%s budget=%s",
            final_tokens,
            token_budget
        )

    return trimmed
